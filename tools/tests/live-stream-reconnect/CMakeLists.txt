cmake_minimum_required(VERSION 3.21)
project(watchtower_qml_reconnect_regression LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Multimedia)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0 gstreamer-video-1.0)
pkg_check_modules(GST_RTSP_SERVER REQUIRED IMPORTED_TARGET gstreamer-rtsp-server-1.0)
set(MODELS "${CMAKE_CURRENT_LIST_DIR}/../../../src/models")
add_executable(qml-stream-reconnect main.cpp "${MODELS}/RtspVideoSource.cpp" "${MODELS}/RtspVideoSource.h")
target_include_directories(qml-stream-reconnect PRIVATE "${MODELS}")
target_link_libraries(qml-stream-reconnect PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Multimedia PkgConfig::GST)
add_executable(stalled-rtsp-fixture rtsp_fixture.c)
target_link_libraries(stalled-rtsp-fixture PRIVATE PkgConfig::GST_RTSP_SERVER)
