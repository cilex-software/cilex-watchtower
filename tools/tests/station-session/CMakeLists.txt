cmake_minimum_required(VERSION 3.21)
project(station_session_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Test)
set(MODELS "${CMAKE_CURRENT_LIST_DIR}/../../../src/models")
add_executable(station-session-test main.cpp "${MODELS}/AuthController.cpp" "${MODELS}/AuthController.h")
target_include_directories(station-session-test PRIVATE "${MODELS}")
target_link_libraries(station-session-test PRIVATE Qt6::Core Qt6::Network Qt6::Test)
enable_testing()
add_test(NAME station-session-contract COMMAND station-session-test)
